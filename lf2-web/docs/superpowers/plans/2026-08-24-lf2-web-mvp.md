# LF2 Web MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the VS-mode-core web remake of Little Fighter 2: 6 original archetypes, 2 stages, up to 4 local humans + CPU bots (max 8 fighters), deterministic sim, Canvas2D rendering.

**Architecture:** Pure deterministic simulation (`stepWorld`) interpreted from JSON frame-data sheets (LF2 `.dat` philosophy); Canvas2D renderer and DOM-overlay UI are read-only consumers fed by `SimEvent`s. Fixed 60Hz accumulator loop; all randomness via seeded PRNG carried in `WorldState`.

**Tech Stack:** TypeScript (strict), Vite, Bun (package manager + test runner), WebAudio, Gamepad API. Zero gameplay dependencies.

**Spec:** `docs/superpowers/specs/2026-08-24-lf2-web-design.md` — executors read both documents. The plan argues from the spec; where this plan names a detail the spec left open, the plan wins.

### Spec refinements resolved here (binding)

1. `Dir` (spec §2.6) = `{ x: -1 | 0 | 1; z: -1 | 0 | 1 }` — x is horizontal intent relative to facing, z is depth-band intent.
2. `stepWorld` returns `StepResult = { state: WorldState; events: SimEvent[] }` (spec §1's loop pseudocode already destructures this form; §2's bare-return wording is superseded).
3. The fixed-timestep loop lives in `src/ui/loop.ts` (pure, testable); `main.ts` composes it. Spec §1 places the clock in `main.ts` — same semantics, extracted for testability.
4. `HitboxDef` gains optional `status?: 'freeze' | 'burn'` (spec §2.2 FSM includes Frozen/Burned states; the causing flag needed a home).
5. Golden-replay headless mode is `HEADLESS=1 bun src/main.ts --replay <file>` per spec §4 testing strategy.

```ts
// src/ui/loop.ts
export interface LoopOpts {
  tickMs: number;
  maxCatchUpMs: number;
  update: () => void;
  render: (alpha: number) => void;
}

export function createLoop(opts: LoopOpts) {
  let acc = 0;
  let prev: number | null = null;
  let running = false;
  let raf = 0;

  function tick(nowMs: number): void {
    if (prev === null) { prev = nowMs; return; }         // prime
    acc += Math.min(nowMs - prev, opts.maxCatchUpMs);
    prev = nowMs;
    while (acc >= opts.tickMs) { opts.update(); acc -= opts.tickMs; }
    opts.render(acc / opts.tickMs);
  }

  function frame(nowMs: number): void {
    if (!running) return;
    tick(nowMs);
    raf = requestAnimationFrame(frame);
  }

  return {
    start() { running = true; prev = null; acc = 0; raf = requestAnimationFrame(frame); },
    stop() { running = false; cancelAnimationFrame(raf); },
    tick,
  };
}
```

```ts
// src/sim/rng.ts — spec §2.5 verbatim
export function mulberry32(seed: number): () => number {
  return function () {
    seed |= 0; seed = (seed + 0x6D2B79F5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```

```js
// scripts/check-determinism.mjs
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join } from "node:path";

const banned = [/Math\.random/, /Date\.now/, /performance\.now/, /from\s+["']node:/, /document\./, /window\./];
let bad = 0;
function scan(dir) {
  for (const entry of readdirSync(dir)) {
    const p = join(dir, entry);
    if (statSync(p).isDirectory()) { scan(p); continue; }
    if (!entry.endsWith(".ts")) continue;
    const lines = readFileSync(p, "utf8").split("\n");
    lines.forEach((line, i) => {
      for (const re of banned) if (re.test(line)) { console.error(`${p}:${i + 1}: ${re} violated: ${line.trim()}`); bad++; }
    });
  }
}
scan(new URL("../src/sim", import.meta.url).pathname);
if (bad > 0) { console.error(`determinism guard: ${bad} violation(s)`); process.exit(1); }
console.log("determinism guard: clean");
```

```ts
// src/sim/inputBuffer.ts
import type { InputFrame } from "./types";

export type Token = "D" | "<" | ">" | "^" | "v" | "A" | "J";

const RING_SIZE = 12;
const LOG_CAP = 24;

export interface PressRecord { token: Token; tick: number }

export interface FighterBuffer {
  push(frame: InputFrame, tick: number): void;
  held(): InputFrame;
  pressLog(): PressRecord[];
  clear(): void;
}

function edges(prev: InputFrame, cur: InputFrame): Token[] {
  const out: Token[] = [];
  if (cur.a && !prev.a) out.push("A");
  if (cur.j && !prev.j) out.push("J");
  if (cur.dHeld && !prev.dHeld) out.push("D");
  if (cur.dir.x === 1 && prev.dir.x !== 1) out.push(">");
  if (cur.dir.x === -1 && prev.dir.x !== -1) out.push("<");
  if (cur.dir.z === -1 && prev.dir.z !== -1) out.push("^");
  if (cur.dir.z === 1 && prev.dir.z !== 1) out.push("v");
  return out;
}

export function createBuffer(size = RING_SIZE): FighterBuffer {
  let ring: InputFrame[] = [];
  let log: PressRecord[] = [];
  return {
    push(cur, tick) {
      const prev = ring.length > 0 ? ring[ring.length - 1]! : { a: false, j: false, dHeld: false, dir: { x: 0, z: 0 } };
      for (const token of edges(prev, cur)) log.unshift({ token, tick });
      if (log.length > LOG_CAP) log.length = LOG_CAP;
      ring.push(cur);
      if (ring.length > size) ring.shift();
    },
    held: () => ring[ring.length - 1] ?? { a: false, j: false, dHeld: false, dir: { x: 0, z: 0 } },
    pressLog: () => log,
    clear() { ring = []; log = []; },
  };
}
```

```ts
// src/sim/specials.ts
import type { Token, PressRecord } from "./inputBuffer";

export function compilePattern(seq: string): Token[] {
  return seq.split("") as Token[];
}

/** Newest-first scan; all tokens in order within windowTicks of nowTick. Longest pattern wins ties. */
export function matchSpecial(
  log: PressRecord[],
  nowTick: number,
  windowTicks: number,
  patterns: Record<string, string>,
): string | null {
  const compiled = Object.entries(patterns)
    .map(([moveId, seq]) => ({ moveId, tokens: compilePattern(seq) }))
    .sort((a, b) => b.tokens.length - a.tokens.length);

  for (const { moveId, tokens } of compiled) {
    let ti = 0;                                   // walk tokens newest→oldest
    for (const rec of log) {
      if (nowTick - rec.tick > windowTicks) break;
      if (rec.token === tokens[ti]) ti++;
      if (ti === tokens.length) return moveId;
    }
  }
  return null;
}
```

```ts
// src/sim/physics.ts
import { GRAVITY, GROUND_FRICTION, AIR_DRAG, ARENA_W, ARENA_D } from "./constants";
import type { Fighter, Projectile, StageRuntime } from "./types";

const DEFAULT_WALLS = { left: 0, right: ARENA_W, restitution: 0.4 };

export function integrateFighter(f: Fighter, _tick: number): void {
  const walls = DEFAULT_WALLS;
  const airborne = f.y < 0;

  if (airborne) {
    f.vy += GRAVITY;
    f.vx *= AIR_DRAG;
  } else {
    f.vy = 0;
    if (f.state !== "walk" && f.state !== "dash" && f.state !== "attack") f.vx *= GROUND_FRICTION;
    else f.vx *= AIR_DRAG;
  }

  f.x += f.vx;
  f.y += f.vy;
  f.z += f.vz;

  // Walls: reflect with restitution
  if (f.x < walls.left) { f.x = walls.left; f.vx = Math.abs(f.vx) * walls.restitution; }
  if (f.x > walls.right) { f.x = walls.right; f.vx = -Math.abs(f.vx) * walls.restitution; }
  // Depth band: hard clamp
  if (f.z < 0) { f.z = 0; f.vz = 0; }
  if (f.z > ARENA_D) { f.z = ARENA_D; f.vz = 0; }
  // Ground clamp
  if (f.y >= 0) { f.y = 0; if (f.vy > 0) f.vy = 0; }
}

export function integrateProjectile(p: Projectile, stage?: StageRuntime): boolean {
  p.ttl -= 1;
  p.vy += p.gravity;
  p.x += p.vx;
  p.y += p.vy;
  const right = stage?.walls.right ?? ARENA_W;
  if (p.ttl <= 0) return false;
  if (p.x < 0 || p.x > right) return false;
  if (p.y > 0) return false;                 // fell past floor
  return true;
}
```

```ts
// src/sim/fighter.ts
import type { CharacterSheet, Fighter, InputFrame, MoveDef, MoveFrame, SimEvent } from "./types";
import { JUMP_IMPULSE, RUN_DASH_SPEED } from "./constants";
import { matchSpecial } from "./specials";

export interface TickWorld { tick: number }

function currentMove(sheet: CharacterSheet, f: Fighter): MoveDef | undefined {
  return f.moveId ? sheet.moves[f.moveId] : undefined;
}

function currentFrame(def: MoveDef | undefined, f: Fighter): { frame: MoveFrame; localTick: number } | undefined {
  if (!def || f.frameIdx === undefined || f.frameTick === undefined) return undefined;
  const fr = def.frames[f.frameIdx];
  if (!fr) return undefined;
  return { frame: fr, localTick: f.frameTick };
}

export function tryStartMove(f: Fighter, moveId: string, sheet: CharacterSheet, world: TickWorld, events: SimEvent[]): boolean {
  const def = sheet.moves[moveId];
  if (!def) return false;
  const cost = def.frames.reduce((acc, fr) => acc + (fr.mpCost ?? 0), 0);   // charged once at start
  if (cost > f.mp) return false;
  f.mp -= cost;
  f.state = "attack"; f.moveId = moveId; f.frameIdx = 0; f.frameTick = 0; f.stateTick = 0;
  f.hitIds.clear();
  const seq = def.sequence;
  if (seq === "D>A") events.push({ type: "castFire", who: f.id });
  return true;
}

function stepMove(f: Fighter, sheet: CharacterSheet, world: TickWorld, events: SimEvent[], input: InputFrame): void {
  const def = currentMove(sheet, f);
  const cur = currentFrame(def, f);
  if (!def || !cur) { f.state = "idle"; return; }

  f.frameTick! += 1;

  // Cancel windows / auto-links
  if (input.a && cur.frame.cancelInto && cur.frame.cancelInto.length > 0) {
    const target = cur.frame.cancelInto[0];
    if (target && tryStartMove(f, target, sheet, world, events)) return;
  }

  const fr = cur.frame;
  if (f.frameTick! >= fr.durationTicks) {
    // advance frame pointer
    if (fr.nextFrameLink) {
      f.moveId = fr.nextFrameLink.moveId; f.frameIdx = fr.nextFrameLink.frameIndex; f.frameTick = 0;
      f.hitIds.clear();
      return;
    }
    f.frameIdx! += 1; f.frameTick = 0;
    if (f.frameIdx! >= def.frames.length) { f.state = "idle"; f.moveId = undefined; f.frameIdx = undefined; f.frameTick = undefined; }
  }
}

function canAct(f: Fighter): boolean {
  return ["idle", "walk", "dash"].includes(f.state);
}

export function advanceFighter(f: Fighter, input: InputFrame, sheet: CharacterSheet, world: TickWorld, events: SimEvent[]): void {
  f.stateTick += 1;
  f.buffer.push(input, world.tick);

  switch (f.state) {
    case "dead": return;

    case "attack": stepMove(f, sheet, world, events, input); break;

    case "hitstun":
      if (f.stateTick >= 16) { f.state = "idle"; f.stateTick = 0; }
      break;

    case "blockstun":
      if (f.stateTick >= 10) { f.state = "idle"; f.stateTick = 0; }
      break;

    case "knockdown":
      if (f.stateTick >= 40) { f.state = "getup"; f.stateTick = 0; f.invulnUntilTick = world.tick + 24; }
      break;

    case "getup":
      if (f.stateTick >= 18) { f.state = "idle"; f.stateTick = 0; }
      break;

    case "grabbed": case "thrown": case "grabAttempt": case "grabbing":
      // resolved by hitdetect/world; passive here
      break;

    case "drinking":
      if (f.stateTick >= 30) { f.state = "idle"; f.stateTick = 0; }
      break;

    default: {                                     // idle/walk/dash/jump
      if (canAct(f)) {
        // 1. specials (buffered sequence)
        const patterns: Record<string, string> = {};
        for (const [id, mv] of Object.entries(sheet.moves)) if (mv.sequence) patterns[id] = mv.sequence;
        const special = matchSpecial(f.buffer.pressLog(), world.tick, 30, patterns);
        if (special && tryStartMove(f, special, sheet, world, events)) return;

        // 2. attack
        if (input.a) { if (tryStartMove(f, "punch1", sheet, world, events)) return; }

        // 3. dash attack by role
        if (f.state === "dash" && input.a) {
          const dashAtk = sheet.roles?.["dashAttack"];
          if (dashAtk && tryStartMove(f, dashAtk, sheet, world, events)) return;
        }

        // 4. jump
        if (input.j) {
          f.state = "jump"; f.stateTick = 0;
          f.vy = sheet.jumpImpulse ?? JUMP_IMPULSE;
          f.vx = input.dir.x * (sheet.walkSpeed ?? 2.2);
          events.push({ type: "jump", who: f.id });
          return;
        }

        // 5. locomotion
        if (input.dir.x !== 0) { f.facing = input.dir.x > 0 ? 1 : -1; f.state = "walk"; }
        else if (f.state === "walk") f.state = "idle";
      }
      break;
    }
  }

  // MP regen (all states except dead)
  if (f.mp < sheet.maxMp) f.mp = Math.min(sheet.maxMp, f.mp + sheet.mpRegenPerTick);
}
```

```ts
// src/sim/hitdetect.ts
import type { CharacterSheet, Fighter, HitboxDef, SimEvent, WorldState } from "./types";
import { COMBO_RESET_TICKS } from "./constants";

interface ResolvedBox { owner: Fighter; box: HitboxDef; ax: number; ay: number; az: number }

function activeBoxes(w: WorldState, sheets: Map<string, CharacterSheet>): ResolvedBox[] {
  const out: ResolvedBox[] = [];
  for (const f of w.fighters) {
    if (f.state !== "attack") continue;
    const sheet = sheets.get(f.charId);
    const mv = f.moveId ? sheet?.moves[f.moveId] : undefined;
    const fr = mv?.frames[f.frameIdx ?? -1];
    if (!fr?.hitbox) continue;
    out.push({ owner: f, box: fr.hitbox, ax: f.x, ay: f.y, az: f.z });
  }
  return out;
}

function overlaps(a: ResolvedBox, f: Fighter): boolean {
  const cx = a.ax + a.box.x * a.owner.facing;
  const cy = a.ay + a.box.y;
  const cz = a.az + a.box.z;
  const fx = f.x, fy = f.y, fz = f.z;
  // victim body approx: standing capsule 28w × 56h × 24d centered on feet origin
  return Math.abs(cx - fx) <= a.box.w / 2 + 14
      && Math.abs(cy - fy - 28) <= a.box.h / 2 + 28
      && Math.abs(cz - fz) <= a.box.d / 2 + 12;
}

export function resolveHits(w: WorldState, sheets: Map<string, CharacterSheet>, events: SimEvent[]): void {
  const boxes = activeBoxes(w, sheets).sort((a, b) =>
    b.box.priority - a.box.priority || a.owner.id - b.owner.id);       // §2.4 step 3

  for (const ab of boxes) {
    const attacker = ab.owner;
    for (const victim of w.fighters) {
      if (victim.id === attacker.id) continue;
      if (attacker.team !== "independent" && victim.team === attacker.team) continue;
      if (victim.state === "dead") continue;
      if (w.tick < victim.invulnUntilTick) continue;                    // §2.4 step 4
      if (!overlaps(ab, victim)) continue;                              // §2.4 step 2
      if (attacker.hitIds.has(victim.id)) continue;

      attacker.hitIds.add(victim.id);
      applyDamage(w, attacker, victim, ab.box, events);                 // §2.4 steps 5–6
    }
  }
}

function applyDamage(w: WorldState, attacker: Fighter, victim: Fighter, box: HitboxDef, events: SimEvent[]): void {
  const dir = attacker.facing;
  const kb = box.knockback;
  const launched = (kb.vy ?? 0) <= -Math.abs(-6.0);                     // ≤ LAUNCH magnitude

  victim.hp -= box.damage;
  victim.vx = kb.vx * dir;
  victim.vy = kb.vy;
  victim.comboCount = attacker.comboCount + 1;                          // combo counts ON victim
  attacker.comboLastTick = w.tick;
  if (w.tick - victim.comboLastTick > COMBO_RESET_TICKS) victim.comboCount = 1;

  events.push({
    type: "hit",
    attacker: attacker.id, victim: victim.id, damage: box.damage,
    heavy: box.type === "heavy", blocked: false,
  });

  if (victim.hp <= 0) {
    victim.state = "dead"; victim.hp = 0;
    events.push({ type: "ko", victim: victim.id });
    checkMatchEnd(w);
    return;
  }
  if (launched) {
    victim.state = "knockdown"; victim.stateTick = 0;
    victim.invulnUntilTick = w.tick + 40 + 18;                          // flight + getup protection
  } else {
    victim.state = "hitstun"; victim.stateTick = 0;
  }
}

function checkMatchEnd(w: WorldState): void {
  const alive = w.fighters.filter((f) => f.state !== "dead");
  if (alive.length === 0) { w.over = true; /* winnerTeam set by caller */ }
}
```

```ts
import { mulberry32 } from "./rng";
import type { Fighter, Pickup, SimEvent, WorldState } from "./types";

export function updateDrops(w: WorldState): void {
  const cfg = w.stage.drops;
  if (w.tick === cfg.firstDropTick || (w.tick > cfg.firstDropTick && (w.tick - cfg.firstDropTick) % cfg.intervalTicks === 0)) {
    const rng = mulberry32(w.rngState);
    // exactly three ordered draws (spec §3.5): weighted pick, x, z
    const entries = Object.entries(cfg.table);
    const totalW = entries.reduce((a, [, wt]) => a + wt, 0);
    let roll = rng() * totalW; let defId = entries[entries.length - 1]![0];
    for (const [id, wt] of entries) { roll -= wt; if (roll <= 0) { defId = id; break; } }
    const x = 80 + rng() * (w.stage.bounds.w - 160);
    const z = rng() * w.stage.bounds.d;
    w.rngState = (w.rngState * 1103515245 + 12345) & 0x7fffffff;   // advance stored stream
    w.pickups.push({ id: w.nextEntityId++, kind: defId.startsWith("milk") || defId === "beer" ? "item" : "weapon",
      defId, x, y: -300, z, vy: 0, spawnedTick: w.tick, shelfLifeTicks: 1200 });
  }
  // shelf life
  w.pickups = w.pickups.filter((p) => w.tick - p.spawnedTick < p.shelfLifeTicks);
}

export function tryPickup(f: Fighter, p: Pickup, _weapons: Record<string, unknown>): boolean {
  if (Math.abs(p.x - f.x) > 24 || Math.abs(p.z - f.z) > 16 || f.y < -20) return false;
  if (f.holdingWeapon) return false;
  f.holdingWeapon = p.defId;
  f.weaponDurability = 3;                            // real durability read in world.ts from weapons table
  return true;
}

export function consume(f: Fighter, items: Record<string, { effect: string; amount: number; consumeTicks: number; shelfLifeTicks: number }>, defId: string, events: SimEvent[]): boolean {
  const it = items[defId];
  if (!it || f.state !== "drinking") return false;
  if (f.stateTick >= it.consumeTicks) {
    if (it.effect === "healHp") f.hp += it.amount;
    else f.mp += it.amount;
    events.push({ type: "itemConsumed", who: f.id, defId });
    return true;
  }
  return false;
}
```

```ts
import type { Fighter, InputFrame, WorldState } from "./types";

interface SimContextMin { sheets: Map<string, unknown>; weapons: Record<string, unknown>; items: Record<string, unknown> }

export function botThink(w: WorldState, self: Fighter, _ctx: SimContextMin): InputFrame {
  const neutral: InputFrame = { a: false, j: false, dHeld: false, dir: { x: 0, z: 0 } };
  if (self.state === "dead" || self.state === "grabbed" || self.state === "thrown") return neutral;

  // emergency dodge: hostile projectile crossing my lane soon?
  for (const p of w.projectiles) {
    if (p.ownerId === self.id) continue;
    const closing = (p.vx > 0) === (self.x > p.x);
    const dist = Math.abs(self.x - p.x);
    if (closing && dist < 140 && Math.abs(p.z - self.z) < 20) {
      return { ...neutral, j: true };                // jump the shot
    }
  }

  if ((self.cooldownUntilTick ?? 0) > w.tick) return neutral;   // decision lock

  const foes = w.fighters.filter((f) => f.id !== self.id && f.state !== "dead"
    && (self.team === "independent" || f.team !== self.team));
  if (foes.length === 0) return neutral;
  const foe = foes.reduce((a, b) => (Math.abs(b.x - self.x) < Math.abs(a.x - self.x) ? b : a));
  const dist = Math.abs(foe.x - self.x);

  const out: InputFrame = { ...neutral, dir: { x: (foe.x > self.x ? 1 : -1), z: 0 } };
  if (dist < 48) { out.a = true; out.dir.x = 0; }
  else if (dist < 160 && self.hp > self.hp * 0.25) { /* keep approaching */ }
  if (self.hp <= self.hp * 0.25 && dist > 96) out.dir.x = -(out.dir.x as 1 | -1);  // retreat
  self.cooldownUntilTick = w.tick + 12;
  return out;
}
```

```ts
import { advanceFighter } from "./fighter";

import { updateDrops } from "./items";
import { botThink } from "./bot";
import type { CharacterSheet, InputFrame, SimEvent, SlotConfig, StageRuntime, WorldState } from "./types";

export interface SimContext {
  sheets: Map<string, CharacterSheet>;
  weapons: Record<string, WeaponDef>;
  items: Record<string, ItemDef>;
}
export interface WeaponDef { kind: string; meleeDamage: number; throwDamage: number; throwVy: number; durability: number; breakOnThrowImpact: boolean; carrierSpeedMul: number; repickupDelayTicks: number }
export interface ItemDef { effect: "healHp" | "restoreMp"; amount: number; consumeTicks: number; shelfLifeTicks: number }
export interface StepResult { state: WorldState; events: SimEvent[] }

export function spawnMatch(cfg: { seed: number; stage: StageRuntime; slots: SlotConfig[]; sheets: Map<string, CharacterSheet> }): WorldState {
  const rng = mulberry32(cfg.seed);
  const fighters = cfg.slots.map((slot, i) => {
    const s = cfg.sheets.get(slot.charId)!;
    return {
      id: i, slot: i, team: slot.team, isBot: !slot.isHuman, charId: slot.charId,
      x: cfg.stage.walls.right / 2 + (i % 2 === 0 ? -1 : 1) * (80 + Math.floor(rng() * 40)),
      y: 0, z: 20 + Math.floor(rng() * (cfg.stage.bounds.d - 40)),
      vx: 0, vy: 0, vz: 0, facing: i % 2 === 0 ? 1 : -1,
      state: "idle" as const, stateTick: 0,
      hp: s.maxHp, mp: s.maxMp, invulnUntilTick: 0,
      hitIds: new Set<number>(), comboCount: 0, comboLastTick: -999,
      buffer: createBuffer(),
    };
  });
  return { tick: 0, seed: cfg.seed, rngState: cfg.seed, stage: cfg.stage, fighters, projectiles: [], pickups: [], nextEntityId: cfg.slots.length, over: false };
}

export function stepWorld(prev: WorldState, inputs: InputFrame[], ctx: SimContext): StepResult {
  const w: WorldState = structuredClone(prev);       // value semantics: caller keeps old state
  w.tick += 1;
  const events: SimEvent[] = [];

  for (let i = 0; i < w.fighters.length; i++) {
    const f = w.fighters[i]!;
    const input = f.isBot ? botThink(w, f, ctx) : inputs[i] ?? NEUTRAL;
    advanceFighter(f, input, ctx.sheets.get(f.charId)!, w, events);
    integrateFighter(f, w.tick);
  }
  resolveHits(w, ctx.sheets, events);
  w.projectiles = w.projectiles.filter((p) => integrateProjectile(p, w.stage));
  updateDrops(w);
  return { state: w, events };
```

```ts
// src/render/renderer.ts (core draw order)
export function draw(w: WorldState, g: CanvasRenderingContext2D, assets: Assets): void {
  g.imageSmoothingEnabled = false;
  drawBackground(g, assets, w.stage.id);
  for (const p of w.pickups) drawPickup(g, p, assets);
  const sprites = [
    ...w.fighters.filter((f) => f.state !== "dead").map((f) => ({ kind: "fighter" as const, f })),
    ...w.projectiles.map((p) => ({ kind: "proj" as const, p })),
  ].sort((a, b) => zOf(a) - zOf(b));               // painter's by ground point
  for (const s of sprites) s.kind === "fighter" ? drawFighter(g, s.f, assets, w) : drawProjectile(g, s.p, assets);
}

function drawFighter(g: CanvasRenderingContext2D, f: Fighter, assets: Assets, w: WorldState): void {
  // resolve current sprite key from sheet move/frame; fallback idle stance
  const sheet = assets.sheets.get(f.charId);
  const mv = f.moveId ? sheet?.moves[f.moveId] : undefined;
  const key = mv?.frames[f.frameIdx ?? 0]?.sprite ?? `${f.charId}_idle`;
  blitSprite(g, assets.fighters, key, f.x, f.y, f.z, f.facing);
  if (f.team !== "independent") drawTeamRing(g, f);           // §4 HUD ring
  drawBars(g, f, w);                                           // HP red / MP blue, top-edge layout handled here
}
```
